import json
import os
from datetime import datetime
from utils.currency import convert, SUPPORTED_CURRENCIES
from utils.validators import validate_signal

BALANCE_FILE = "data/paper_balance.json"
POSITIONS_FILE = "data/positions.json"
TRADES_LOG = "logs/paper_trades.log"
SUMMARY_FILE = "data/paper_summary.json"

DEFAULT_BALANCES = {
    "USD": {"cash": 1000},
    "ZAR": {"cash": 18500}
}


def load_json(file_path, default=None):
    if not os.path.exists(file_path):
        return default if default is not None else {}
    with open(file_path, "r") as f:
        try:
            return json.load(f)
        except json.JSONDecodeError:
            return default if default is not None else {}


def save_json(file_path, data):
    with open(file_path, "w") as f:
        json.dump(data, f, indent=4)


def log_trade(message):
    timestamp = datetime.utcnow().isoformat()
    with open(TRADES_LOG, "a") as f:
        f.write(f"{timestamp} | {message}\n")


def validate_balance(balances, currency):
    if currency not in balances:
        raise ValueError(f"Missing balance for {currency}")
    if balances[currency]["cash"] < 0:
        raise ValueError(f"Negative balance for {currency}")


def validate_position(position):
    required_keys = {"entry_price", "entry_time", "currency", "trade_size"}
    if not required_keys.issubset(position.keys()):
        raise ValueError("Corrupt position data")


def update_summary(coin, pnl):
    summary = load_json(SUMMARY_FILE, default={})
    coin_summary = summary.get(coin, {"trades": 0, "total_pnl": 0.0})
    coin_summary["trades"] += 1
    coin_summary["total_pnl"] += pnl
    summary[coin] = coin_summary

    # Update total PnL across all coins
    total_pnl = sum(c["total_pnl"] for c in summary.values())
    summary["total_pnl"] = total_pnl

    save_json(SUMMARY_FILE, summary)
    return summary


def execute_paper_trade(coin, signal, price, currency="USD", trade_size=100):
    """
    Executes a paper trade with multi-currency support and PnL summary tracking.

    signal: 1 = BUY, -1 = SELL, 0 = HOLD
    """
    balances = load_json(BALANCE_FILE, default=DEFAULT_BALANCES.copy())
    positions = load_json(POSITIONS_FILE, default={})

    # Ensure currency exists
    if currency not in balances:
        balances[currency] = {"cash": 0}

    account = balances[currency]
    position = positions.get(coin)
    timestamp = datetime.utcnow().isoformat()

    # --- BUY ---
    if signal == 1 and position is None:
        if account["cash"] < trade_size:
            return f"{coin} NOT ENOUGH {currency} BALANCE"

        account["cash"] -= trade_size
        positions[coin] = {
            "entry_price": price,
            "entry_time": timestamp,
            "currency": currency,
            "trade_size": trade_size
        }

        save_json(POSITIONS_FILE, positions)
        save_json(BALANCE_FILE, balances)

        log_trade(f"BUY | {coin} | {price} | {trade_size} {currency}")
        return f"{coin} PAPER BUY @ {price} ({currency})"

    # --- SELL ---
    if signal == -1 and position is not None:
        entry_price = position["entry_price"]
        trade_size = position["trade_size"]
        currency = position["currency"]

        pnl = (price - entry_price) * (trade_size / entry_price)
        balances[currency]["cash"] += trade_size + pnl

        # Update summary
        summary = update_summary(coin, pnl)

        log_trade(f"SELL | {coin} | {price} | PnL: {pnl:.2f} {currency}")

        del positions[coin]
        save_json(POSITIONS_FILE, positions)
        save_json(BALANCE_FILE, balances)

        return f"{coin} PAPER SELL @ {price} | PnL: {pnl:.2f} {currency}"

    # HOLD or no action
    return None


def get_paper_summary():
    """
    Returns current summary for all coins and total PnL.
    """
    summary = load_json(SUMMARY_FILE, default={})
    if not summary:
        return "No trades yet."
    lines = []
    for coin, data in summary.items():
        if coin == "total_pnl":
            continue
        lines.append(f"{coin}: Trades={data['trades']}, Total PnL={data['total_pnl']:.2f}")
    lines.append(f"TOTAL PnL: {summary.get('total_pnl', 0.0):.2f}")
    return "\n".join(lines)
