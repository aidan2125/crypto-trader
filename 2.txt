Step 1: Create Run Script
bashcd ~/crypto-trader

# Create a simple run script
cat > scripts/run_trader.sh << 'EOF'
#!/bin/bash
# Simple script to run the trading bot

cd /home/aidan/crypto-trader
source venv/bin/activate
python3 main_enhanced.py
EOF

# Make it executable
chmod +x scripts/run_trader.sh

# Test it works
./scripts/run_trader.sh

Step 2: Create systemd Service
This makes your bot run automatically:
bash# Create the service file
mkdir -p ~/.config/systemd/user

cat > ~/.config/systemd/user/crypto-trader.service << 'EOF'
[Unit]
Description=Crypto Trading Bot (Paper Trading)
After=network.target

[Service]
Type=oneshot
WorkingDirectory=/home/aidan/crypto-trader
EnvironmentFile=/home/aidan/crypto-trader/.env
ExecStart=/home/aidan/crypto-trader/scripts/run_trader.sh
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=default.target
EOF

echo " Service file created"

Step 3: Create systemd Timer
This tells the bot to run every 5 minutes:
bashcat > ~/.config/systemd/user/crypto-trader.timer << 'EOF'
[Unit]
Description=Run Crypto Trading Bot every 5 minutes

[Timer]
OnBootSec=2min
OnUnitActiveSec=5min
Persistent=true

[Install]
WantedBy=timers.target
EOF

echo " Timer file created"

Step 4: Enable and Start
bash# Reload systemd
systemctl --user daemon-reload

# Enable the timer (starts on boot)
systemctl --user enable crypto-trader.timer

# Start the timer NOW
systemctl --user start crypto-trader.timer

# Check status
systemctl --user status crypto-trader.timer
